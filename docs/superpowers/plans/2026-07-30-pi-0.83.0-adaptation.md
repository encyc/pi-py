**Goal:** Adapt the existing Python SDK surface to `earendil-works/pi@v0.83.0`, including pending/raw stop-reason semantics, injectable HTTP clients, the malformed OpenAI tool-delta fix, synchronized package versions, and live DeepSeek/OpenCode Zen verification.

**Architecture:** Keep the repository's SDK-only boundary. Leave OAuth, CLI/TUI, extensions, images, and unsupported providers unchanged.

**Tech Stack:** Python 3.11+, asyncio, Pydantic v2, OpenAI Python SDK, Anthropic Python SDK, pytest/pytest-asyncio, Ruff, Mypy, uv workspace.

class _AsyncItems:
    def __init__(self, items):
        self._items = iter(items)

    def __aiter__(self):
        return self

    async def __anext__(self):
        try:
            return next(self._items)
        except StopIteration as exc:
            raise StopAsyncIteration from exc


def _chunk(*, finish_reason=None, delta=None):
    delta = delta or SimpleNamespace(content=None, tool_calls=None)
    choice = SimpleNamespace(finish_reason=finish_reason, delta=delta)
    return SimpleNamespace(usage=None, choices=[choice])


def _fake_openai_client(chunks):
    async def create(**kwargs):
        return _AsyncItems(chunks)

    completions = SimpleNamespace(create=create)
    return SimpleNamespace(chat=SimpleNamespace(completions=completions))


async def _collect_openai(monkeypatch, chunks, *, context=None, options=None):
    monkeypatch.setattr(
        openai_provider,
        "_create_client",
        lambda model, api_key, headers, http_client=None: _fake_openai_client(chunks),

        _make_model(),

    events = []
    async for event in stream:

        if isinstance(event, StartEvent):

    return events, start_reasons, await stream.result()
